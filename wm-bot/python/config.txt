#   本文件放在 python/ 根目录，是给别人照着改的样例，会被提交到仓库。
#
# 正确用法（三步）：
#   1. 把本文件复制一份到同目录的 data/ 文件夹下，文件名仍是 config.txt，
#      即最终路径 python/data/config.txt（没有 data/ 文件夹就自己新建一个）。
#   2. 用记事本 / 手机文本编辑器打开 data/config.txt，把 WM_QQ_APPID、
#      WM_QQ_SECRET 填成你自己在 q.qq.com 申请的凭证；其它项按需修改，
#      每一项下面都有中文注释。
#   3. 保存后运行 python qq_bot.py。程序【只读取 data/config.txt】，
#      不会读这份模板；只改模板却没有复制到 data/ 是不会生效的。
#
# 关于上传 GitHub：项目根目录已放好 .gitignore，其中 data/ 一行会把
#   data/config.txt（密钥）、data/groups.txt（群白名单）、缓存与日志
#   全部排除。不要手动把 data/ 加进版本控制，就不会泄露任何个人信息。
#   data/groups.txt 也要自己在 data/ 下新建（每行一个 group_openid）；
#   不建也行 —— 默认不监控任何群，私聊照常可用。
#
# 写法：每行一个「名称 = 值」
#   - 「#」后面的文字都是注释，随便写，不影响运行；
#   - 值里不要出现「#」；留空 = 用程序内置默认值；
#   - 等号左边的名称不要改；名称大小写不敏感；
#   - 路径一律相对【python 脚本所在目录】（不是相对本配置文件），
#     不要写盘符、不要写 ../ 上层路径；
#   - 编码 UTF-8（Windows 记事本存成带 BOM 的也能读）。

# 官方接口根地址，一般不用改
WM_QQ_API_BASE = https://api.bot.qq.com

# 事件订阅位掩码，一般不用改（33554432 = 群@消息 + 单聊消息）
WM_QQ_INTENTS = 33554432

# ---------- 二、群白名单（只监控写进文件的群） ----------

# 白名单文件：每行一个 group_openid；只有写在里面的群才回复，其它群一律静默。
# openid 是「CC2129F9E...」这种字符串，不是 QQ 里显示的数字群号。
# 路径相对 python 脚本所在目录；请在 data/ 下自己新建 groups.txt。
WM_QQ_GROUPS_FILE = data/groups.txt

# 调试开关：0 = 只监控白名单群（正式运行必须是 0）
#           1 = 临时监控所有群，仅用于第一次抓群 openid，抓完立刻改回 0
# 也可临时运行 python qq_bot.py --allow-all-groups 开启一次，不必改这里。
WM_QQ_ALLOW_ALL_GROUPS = 0

# ---------- 三、查询与缓存 ----------

# 缓存新鲜度，单位【分钟】，默认 60。想改成 15 分钟，就把 60 改成 15。
# 用户发起查询时才更新：缓存没超过这个间隔 -> 直接秒回（不联网）；
#                       超过这个间隔       -> 先联网刷新缓存、再回复用户。
# 绝不把几天 / 几个月前的旧缓存当结果发出去；刷新失败会明确报错，
# 不拿旧数据顶包。
WM_CACHE_MINUTES = 60

# 数据源根地址。默认官方 API，一般不用改。
# 如果你的网络访问不了 api.warframe.market，可把这里改成可用的代理 / 镜像地址。
WM_API_BASE = https://api.warframe.market

# 单次联网请求的超时秒数，默认 15。连不上就立刻明确报错，不会让机器人卡住不回复。
WM_HTTP_TIMEOUT = 15

# ---------- 四、OneBot 后端专用（用官方后端时下面几行用不上） ----------

# 连接模式：client = 主动连过去；server = 等对方连进来（手机挂机常用）

# ---------- 五、固定规则（不提供配置，写在这里只是说明） ----------
# 同域名访问间隔 3~5 秒：硬规则，真实源与离线演示都真等，不允许调小。
# 刷新失败：用户这次只尝试 1 次并立刻收到明确报错（不发旧数据），
#           1 / 3 / 7 分钟重试（累计最多 4 次）在后台继续，成功后自动更新缓存；
#           这样真实源挂掉时机器人不会假死、不会错过 QQ 被动回复的 5 分钟窗口。
# 缓存只按「用户实际查过的物品」建立，绝不预拉全站物品清单。
